const DATE_FORMATS = {
  de: { weekday: 'short', day: 'numeric', month: 'long', year: 'numeric' },
  en: { weekday: 'short', day: 'numeric', month: 'long', year: 'numeric' },
  fr: { weekday: 'short', day: 'numeric', month: 'long', year: 'numeric' },
  nl: { weekday: 'short', day: 'numeric', month: 'long', year: 'numeric' }
};

function getInitialLanguage() {
  const params = new URLSearchParams(window.location.search);
  const fromQuery = params.get('lang');
  if (fromQuery && translations[fromQuery]) return fromQuery;
  const saved = localStorage.getItem('siteLanguage');
  if (saved && translations[saved]) return saved;
  const browser = (navigator.language || 'en').slice(0, 2).toLowerCase();
  return translations[browser] ? browser : 'en';
}

function formatSingleDate(dateString, lang) {
  const date = new Date(`${dateString}T12:00:00`);
  return new Intl.DateTimeFormat(lang, DATE_FORMATS[lang] || DATE_FORMATS.en).format(date);
}

function formatRangeDate(startString, endString, lang) {
  const start = new Date(`${startString}T12:00:00`);
  const end = new Date(`${endString}T12:00:00`);
  const sameMonth = start.getMonth() === end.getMonth() && start.getFullYear() === end.getFullYear();
  if (lang === 'de') {
    return sameMonth
      ? `${start.getDate()}–${end.getDate()}. ${new Intl.DateTimeFormat('de', { month: 'long', year: 'numeric' }).format(start)}`
      : `${formatSingleDate(startString, lang)} – ${formatSingleDate(endString, lang)}`;
  }
  if (lang === 'fr') {
    return sameMonth
      ? `${start.getDate()}–${end.getDate()} ${new Intl.DateTimeFormat('fr', { month: 'long', year: 'numeric' }).format(start)}`
      : `${formatSingleDate(startString, lang)} – ${formatSingleDate(endString, lang)}`;
  }
